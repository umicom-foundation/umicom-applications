2. Run the focused Windows checks

This builds the real affected C sources without requiring the entire GUI dependency graph. Expect 6 naming/learning tests and 14 input-validation tests. Checks and setup remain active in Release configurations too.

Set-Location "C:\umicom\Umicom-Applications" -ErrorAction Stop
$env:Path = "C:\msys64\ucrt64\bin;$env:Path"

foreach ($suite in @("release_qualification", "input_validation")) {
    $buildDir = ".\build\windows-ucrt64-$suite"
    & "C:\msys64\ucrt64\bin\cmake.exe" `
        -S ".\framework\tests\$suite" `
        -B $buildDir `
        -G Ninja `
        -DCMAKE_BUILD_TYPE=Debug `
        -DCMAKE_C_COMPILER="C:/msys64/ucrt64/bin/gcc.exe" `
        -DCMAKE_MAKE_PROGRAM="C:/msys64/ucrt64/bin/ninja.exe"
    if ($LASTEXITCODE -ne 0) { throw "$suite configuration failed." }

    & "C:\msys64\ucrt64\bin\cmake.exe" --build $buildDir --parallel 2
    if ($LASTEXITCODE -ne 0) { throw "$suite build failed." }

    & "C:\msys64\ucrt64\bin\ctest.exe" `
        --test-dir $buildDir --no-tests=error --output-on-failure
    if ($LASTEXITCODE -ne 0) { throw "$suite tests failed." }
}
