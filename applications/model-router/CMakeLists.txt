cmake_minimum_required(VERSION 3.25)
project(UmicomModelRouter LANGUAGES C)
include(CTest)
set(UMICOM_FRAMEWORK_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../framework" CACHE PATH "Canonical Framework checkout")
option(UMICOM_WORKER_ENABLE_TEXT "Use the canonical Framework local chat provider" ON)
option(UMICOM_MODEL_ROUTER_DESKTOP "Build the existing Framework GTK desktop host from its SDK" OFF)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WORKER_CURL REQUIRED IMPORTED_TARGET libcurl>=7.85)
pkg_check_modules(WORKER_JSONC REQUIRED IMPORTED_TARGET json-c>=0.15)
if(NOT EXISTS "${UMICOM_FRAMEWORK_SOURCE_DIR}/include/umicom/ai_workspace/worker_transport.h")
    message(FATAL_ERROR "Apply the Framework worker_transport additions to the canonical checkout first.")
endif()
include("${UMICOM_FRAMEWORK_SOURCE_DIR}/cmake/UmicomAiWorkerTransport.cmake")
include("${UMICOM_FRAMEWORK_SOURCE_DIR}/cmake/UmicomAiStoryboardBridge.cmake")
add_executable(umicom-ai-bridge src/main.c)
target_link_libraries(umicom-ai-bridge PRIVATE Umicom::ai_worker_transport Umicom::ai_storyboard_bridge PkgConfig::WORKER_JSONC)
if(UMICOM_WORKER_ENABLE_TEXT)
    if(TARGET Umicom::ai_workspace)
        target_link_libraries(umicom-ai-bridge PRIVATE Umicom::ai_workspace)
    else()
        # This is the SAME provider source, not an alternate implementation.
        set(_canonical_sources cancellation.c text.c local_chat_codec.c local_chat_provider.c)
        set(_source_paths "")
        foreach(_name IN LISTS _canonical_sources)
            set(_path "${UMICOM_FRAMEWORK_SOURCE_DIR}/src/ai_workspace/${_name}")
            if(NOT EXISTS "${_path}")
                message(FATAL_ERROR "Required canonical source missing: ${_path}")
            endif()
            list(APPEND _source_paths "${_path}")
        endforeach()
        add_library(umicom_router_existing_provider STATIC ${_source_paths})
        target_include_directories(umicom_router_existing_provider PUBLIC "${UMICOM_FRAMEWORK_SOURCE_DIR}/include")
        target_compile_definitions(umicom_router_existing_provider PRIVATE UMICOM_AI_WORKSPACE_HAS_HTTP=1)
        target_link_libraries(umicom_router_existing_provider PRIVATE PkgConfig::WORKER_CURL PkgConfig::WORKER_JSONC)
        target_link_libraries(umicom-ai-bridge PRIVATE umicom_router_existing_provider)
    endif()
    target_compile_definitions(umicom-ai-bridge PRIVATE UMICOM_WORKER_HAS_TEXT=1)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(umicom-ai-bridge PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
if(UMICOM_MODEL_ROUTER_DESKTOP)
    if(NOT TARGET Umicom::ai_workspace_gtk4)
        find_package(UmicomFramework CONFIG REQUIRED)
    endif()
    if(NOT TARGET Umicom::ai_workspace_gtk4)
        message(FATAL_ERROR "Desktop needs the installed Framework GTK workspace target. It is not replaced by a browser wrapper.")
    endif()
    add_executable(umicom-model-router-desktop src/desktop.c)
    target_link_libraries(umicom-model-router-desktop PRIVATE Umicom::ai_workspace_gtk4)
    install(TARGETS umicom-model-router-desktop RUNTIME DESTINATION tools/native)
endif()
install(TARGETS umicom-ai-bridge RUNTIME DESTINATION tools/native)

# Stage non-system DLLs at both build and install locations. This leaves the
# existing executable name, provider links and optional desktop target unchanged.
option(UMICOM_MODEL_ROUTER_STAGE_RUNTIME "Stage coherent Windows helper DLLs" ON)
if(WIN32 AND UMICOM_MODEL_ROUTER_STAGE_RUNTIME)
    get_filename_component(_router_runtime_bin "${CMAKE_C_COMPILER}" DIRECTORY)
    set(UMICOM_MODEL_ROUTER_RUNTIME_DIR "${_router_runtime_bin}" CACHE PATH "Runtime bin directory used to link the bridge")
    include("${UMICOM_FRAMEWORK_SOURCE_DIR}/cmake/UmicomNativeRuntime.cmake")
    umicom_prepare_native_runtime(TARGET umicom-ai-bridge
        RUNTIME_DIR "${UMICOM_MODEL_ROUTER_RUNTIME_DIR}" DESTINATION tools/native)
endif()
